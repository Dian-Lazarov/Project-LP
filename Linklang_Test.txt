core {
  int alpha, beta;
  shared count;
  sem mutex;


  alpha = 10;
  beta = alpha + 20 * 2;


  xin (alpha);
  xout ("Result is:");
  xout (beta);


  if (alpha) {
    beta = beta - 1;
  } else {
    beta = beta + 1;
  }


  while (beta) {
    beta = beta - 1;
  }


  for (alpha = 0 ; alpha ; alpha = alpha + 1) {
    xout (alpha);
  }


  init (mutex , 1);
  wait (mutex);
  count = count + 1;
  signal (mutex);
}
