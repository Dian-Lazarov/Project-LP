core {
    int n, i, is_prime;
    int quotient, remainder;

    xout(#Въведете число за проверка: #);
    xin(n);

    if (n <= 1) {
        xout(#Числото не е просто.#);
    } else {
        is_prime = 1;  ~~ Приемаме, че числото е просто (1 = true)
        i = 2;

        ~~ Цикъл с предсрочен изход (спира, ако is_prime стане 0)
        while (i < n and is_prime @= 1) {
            
            ~~ Изчисляване на остатък от деление: n % i
            quotient = n / i;
            remainder = n - quotient * i;

            if (remainder @= 0) {
                ~~ Открит е делител! Сваляме флага и цикълът ще прекъсне предсрочно.
                is_prime = 0;
            } else {
                i = i + 1;
            }
        }

        ~~ Проверка на резултата след края на цикъла
        if (is_prime @= 1) {
            xout(#Числото е просто.#);
        } else {
            xout(#Числото не е просто.#);
        }
    }
};
