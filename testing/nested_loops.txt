core {
    ~~ 1. Всички декларации са най-отгоре
    int i, j, result;
    int threshold;

    threshold = 4;
    xout(#--- Тест на вложени цикли и условен оператор ---#);

    i = 1;
    
    ~~ 2. Външен цикъл (while)
    while (i <= 3) {
        xout(#Външен цикъл, i = #);
        xout(i);

        ~~ 3. Вътрешен цикъл (for)
        for (j = 1; j <= 3; j = j + 1) {
            result = i * j;
            
            ~~ 4. Условен оператор (if-else), вложен в цикъла
            if (result > threshold) {
                xout(#  -> Резултатът е НАД прага: #);
                xout(result);
            } else {
                xout(#  -> Резултатът е под/равен на прага: #);
                xout(result);
            }
        }

        i = i + 1;
    }
    
    xout(#--- Край на теста ---#);
};
