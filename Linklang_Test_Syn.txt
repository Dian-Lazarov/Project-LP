core {
    int test;
    ~~ Декларационна част (задължително в началото)
    int a, b, result;
    int counter, max_val;
    test = 7;

    /~ 
       Тестване на входно-изходни операции
       и инициализация на променливи
    ~/
    xout(#Въведете стойност за a: #);
    xin(a);
    xout(#Въведете стойност за b: #);
    xin(b);

    max_val = 100;
    counter = 0;

    ~~ Тестване на условен оператор, сложни изрази и логически връзки
    if (a @= b and (a + b) < max_val) {
        ~~ Проверка на приоритетите: умножение, събиране, скоби
        result = a * 2 + b / 3 - (a - b);
        xout(#Условието е изпълнено! Резултат: #);
        xout(result);
    } else {
        result = 0;
        xout(#Условието не е изпълнено.#);
    }

    ~~ Тестване на цикъл while и оператор за равенство (?=)
    while (counter < 5) {
        if (counter ?= 3) {
            xout(#Броячът достигна 3!#);
        }
        counter = counter + 1;
    }

    ~~ Тестване на цикъл for със сложна стъпка
    for (a = 0; a <= 10; a = a + 2) {
        xout(a);
    }
};
